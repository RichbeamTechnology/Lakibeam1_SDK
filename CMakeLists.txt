cmake_minimum_required(VERSION 3.15)
project(LakiBeamCapture VERSION 1.0.0 LANGUAGES CXX)

# Statically link the MSVC runtime so the customer does not need a VC++ runtime installer.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(lakibeam_capture offline/LakiBeamCapture.cpp)
target_include_directories(lakibeam_capture PUBLIC offline)
target_compile_features(lakibeam_capture PUBLIC cxx_std_17)
add_executable(LakiBeamCapture offline/main.cpp)
target_link_libraries(LakiBeamCapture PRIVATE lakibeam_capture)
if(WIN32)
  target_link_libraries(LakiBeamCapture PRIVATE comdlg32 user32)
endif()
if(MSVC)
  target_compile_options(lakibeam_capture PRIVATE /W4 /utf-8)
  target_compile_options(LakiBeamCapture PRIVATE /W4 /utf-8)
else()
  target_compile_options(lakibeam_capture PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(LakiBeamCapture PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME capture_regression COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_capture.py" "$<TARGET_FILE:LakiBeamCapture>")
  endif()
endif()
